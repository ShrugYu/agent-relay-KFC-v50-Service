快速开始

git clone https://github.com/ShrugYu/agent-relay-KFC-v50-Service.git
cd agent-relay-KFC-v50-Service
python3 openai_crazy_thursday.py      # Python 版
# 或
go build -o kfcgo . && ./kfcgo        # Go 版

# 默认监听 127.0.0.1:8788
curl http://127.0.0.1:8788/v1/chat/completions \
  -H "Content-Type: application/json" \
  -d '{"model":"gpt-5.5","messages":[{"role":"user","content":"你好"}]}'
